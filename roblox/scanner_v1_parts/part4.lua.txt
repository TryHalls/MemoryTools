        lines[#lines + 1] = string.format("[%s] %s", item.className, item.path)
        lines[#lines + 1] = "DirectChildren: " .. tostring(item.childCount)
        if next(item.childClasses) ~= nil then
            lines[#lines + 1] = "ChildClasses: " .. formatClassCounts(item.childClasses)
        end
        lines[#lines + 1] = ""
    end

    lines[#lines + 1] = "[ATTRIBUTES / TAGS]"
    table.sort(state.metadata, function(a, b) return a.path < b.path end)
    for _, item in ipairs(state.metadata) do
        lines[#lines + 1] = string.format("[%s] %s", item.className, item.path)
        appendMetadataLines(lines, item)
        lines[#lines + 1] = ""
    end

    lines[#lines + 1] = "[VALUES]"
    table.sort(state.values, function(a, b) return a.path < b.path end)
    for _, item in ipairs(state.values) do
        lines[#lines + 1] = string.format("[%s] %s = %s", item.className, item.path, serialize(item.value))
    end

    lines[#lines + 1] = ""
    lines[#lines + 1] = "[TOOLS]"
    table.sort(state.tools, function(a, b) return a.path < b.path end)
    for _, item in ipairs(state.tools) do
        lines[#lines + 1] = "Path: " .. item.path
        lines[#lines + 1] = "ToolTip: " .. sanitize(item.tooltip)
        lines[#lines + 1] = "RequiresHandle: " .. tostring(item.requiresHandle)
        lines[#lines + 1] = "CanBeDropped: " .. tostring(item.canBeDropped)
        appendMetadataLines(lines, item)
        lines[#lines + 1] = ""
    end

    lines[#lines + 1] = "[INTERACTIONS]"
    table.sort(state.interactions, function(a, b) return a.path < b.path end)
    for _, item in ipairs(state.interactions) do
        lines[#lines + 1] = string.format("[%s] %s", item.className, item.path)
        if item.actionText ~= nil then lines[#lines + 1] = "ActionText: " .. sanitize(item.actionText) end
        if item.objectText ~= nil then lines[#lines + 1] = "ObjectText: " .. sanitize(item.objectText) end
        if item.maxDistance ~= nil then lines[#lines + 1] = "MaxDistance: " .. tostring(item.maxDistance) end
        if item.holdDuration ~= nil then lines[#lines + 1] = "HoldDuration: " .. tostring(item.holdDuration) end
        lines[#lines + 1] = ""
    end

    lines[#lines + 1] = "[GUI ENTRYPOINTS / INTERACTIVE CONTROLS]"
    table.sort(state.gui, function(a, b) return a.path < b.path end)
    for _, item in ipairs(state.gui) do
        lines[#lines + 1] = string.format("[%s] %s", item.className, item.path)
        if item.text and item.text ~= "" then
            lines[#lines + 1] = "Text: " .. sanitize(item.text)
        end
    end

    return table.concat(lines, "\n")
end

local function splitText(text, maxLength)
    local chunks = {}
    local cursor = 1
    local totalLength = #text

    while cursor <= totalLength do
        local finish = math.min(cursor + maxLength - 1, totalLength)
        if finish < totalLength then
            local candidate = string.sub(text, cursor, finish)
            local lastNewline = nil
            for index = #candidate, 1, -1 do
                if string.sub(candidate, index, index) == "\n" then
                    lastNewline = index
                    break
                end
            end
            if lastNewline and lastNewline > math.floor(maxLength * 0.65) then
                finish = cursor + lastNewline - 2
            end
        end

        chunks[#chunks + 1] = string.sub(text, cursor, finish)
        cursor = finish + 1
    end

    if #chunks == 0 then
        chunks[1] = ""
    end
    return chunks
end

local function clipboardFunction()
    if type(setclipboard) == "function" then
        return setclipboard
    end
    if type(toclipboard) == "function" then
        return toclipboard
    end
    return nil
end

local function copyActiveChunk()
    if not activeChunks then
        return
    end

    local copy = clipboardFunction()
    if not copy then
        setStatus("Portapapeles no disponible. El contenido se mostró en terminal.")
        terminalSet(activeChunks[activeChunkIndex])
        return
    end

    local ok, err = pcall(function()
        copy(activeChunks[activeChunkIndex])
    end)

    if ok then
        local label = string.format("%s • parte %d/%d copiada", activeExportName or "Reporte", activeChunkIndex, #activeChunks)
        setStatus(label)
        terminalSet(
            "[COPIADO]\n" .. label .. "\n\n" ..
            "Caracteres en esta parte: " .. tostring(#activeChunks[activeChunkIndex]) .. "\n" ..
            "Usa SIGUIENTE PARTE si hay más bloques."
        )
        NextPart.Visible = #activeChunks > 1
        if #activeChunks > 1 then
            NextPart.Text = string.format("SIGUIENTE PARTE (%d/%d)", activeChunkIndex, #activeChunks)
        end
    else
        setStatus("Error copiando al portapapeles")
        terminalSet("[ERROR] No se pudo copiar:\n" .. tostring(err))
    end
end

local function exportCategory(name, text)
    if not reportData then
        return
    end
    activeExportName = name
    activeChunks = splitText(text, CONFIG.CHUNK_SIZE)
    activeChunkIndex = 1
    copyActiveChunk()
end

    end
    copyActiveChunk()
end)
