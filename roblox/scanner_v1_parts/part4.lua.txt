                out[#out + 1] = rendered
                if #out >= CONFIG.MAX_INTERESTING_LINES then break end
            end
        end
    end
    return out
end

local function extractCallSites(lines)
    local calls = {}
    for lineNumber, line in ipairs(lines) do
        for _, token in ipairs(NETWORK_TOKENS) do
            if string.find(line, token, 1, true) then
                local receiver = callReceiver(line, token)
                local context = contextBlock(lines, lineNumber)
                local literals = extractQuotedLiterals(context)
                calls[#calls + 1] = {
                    token = token,
                    line = lineNumber,
                    receiver = receiver,
                    aliasOrigin = findAliasOrigin(lines, lineNumber, receiver),
                    literals = literals,
                    context = context,
                }
                if #calls >= CONFIG.MAX_CALLS_PER_SCRIPT then
                    return calls
                end
            end
        end
    end
    return calls
end

local function analyseTarget(target)
    local record = target.record
    local ok, sourceOrError = pcall(function() return decompile(record.instance) end)
    if not ok or type(sourceOrError) ~= "string" or sourceOrError == "" then
        state.source.failures[#state.source.failures + 1] = target.label .. " :: " .. sanitize(sourceOrError or "no source")
        return nil
    end
    local source = sourceOrError
    state.source.sourceChars = state.source.sourceChars + #source
    local lines = splitLines(source)
    local requires = extractRequireLines(lines)
    local calls = extractCallSites(lines)
    local interesting = extractInterestingLines(lines)
    state.source.totalRequires = state.source.totalRequires + #requires
    state.source.totalCalls = state.source.totalCalls + #calls
    if target.label == "Networking" or target.label == "Remotes" then
        state.source.coreSource[target.label] = source
    end
    return {
        label = target.label,
        className = record.className,
        path = record.path,
        sourceChars = #source,
        lineCount = #lines,
        requires = requires,
        calls = calls,
        interesting = interesting,
    }
end

local function runNetworkSourceAnalysis()
    state.source = emptySourceState()
    state.source.attempted = true
    if type(decompile) ~= "function" then
        state.source.failures[#state.source.failures + 1] = "decompile() unavailable"
        return false, "decompile() no está disponible"
    end
    local targets = selectTargets()
    terminalSet("[NETWORK SOURCE] V1.5\n[INFO] Solo análisis estático de módulos objetivo. No require(), no remotes, no getconnections().")
    for index, target in ipairs(targets) do
        setStatus(string.format("NETWORK SOURCE %d/%d • %s", index, #targets, target.label), #targets > 0 and (index - 1) / #targets or 0)
        terminalAdd(string.format("[SOURCE] %d/%d %s", index, #targets, target.record.path))
        local result = analyseTarget(target)
        if result then state.source.results[#state.source.results + 1] = result end
        task.wait()
    end
    setStatus("NETWORK SOURCE completado", 1)
    return true
end
