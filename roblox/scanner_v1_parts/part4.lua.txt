end

local function extractFocusContexts(lines)
    local out, covered = {}, {}
    for i,line in ipairs(lines) do
        local terms = focusMatches(line)
        if #terms > 0 and not covered[i] then
            local context, first, last = contextBlock(lines, i)
            out[#out+1] = {
                line=i,
                terms=terms,
                context=context,
            }
            for n=first,last do covered[n] = true end
            if #out >= CONFIG.MAX_CONTEXTS_PER_TARGET then break end
        end
    end
    return out
end

local function analyseTarget(target)
    local ok, source = pcall(function() return decompile(target.record.instance) end)
    if not ok or type(source) ~= "string" or source == "" then
        state.source.failures[#state.source.failures+1] = target.label .. " :: " .. sanitize(source or "no source")
        return nil
    end
    state.source.sourceChars = state.source.sourceChars + #source
    local lines = splitLines(source)
    local contexts = extractFocusContexts(lines)
    state.source.contexts = state.source.contexts + #contexts
    return {
        label=target.label,
        path=target.record.path,
        className=target.record.className,
        chars=#source,
        lines=#lines,
        requires=extractRequires(lines),
        contexts=contexts,
    }
end

local function runSource()
    state.source = emptySource()
    state.source.attempted = true
    if type(decompile) ~= "function" then
        state.source.failures[#state.source.failures+1] = "decompile() unavailable"
        return false, "decompile() no disponible"
    end
    local targets = selectTargets()
    terminalSet("[FINAL SOURCE] V1.6\n[INFO] Decompilando solo targets de Auto Steal + Player.")
    for index,target in ipairs(targets) do
        setStatus(string.format("SOURCE %d/%d • %s", index, #targets, target.label), #targets > 0 and (index-1)/#targets or 0)
        terminalAdd(string.format("[SOURCE] %d/%d %s", index, #targets, target.record.path))
        local result = analyseTarget(target)
        if result then state.source.results[#state.source.results+1] = result end
        task.wait()
    end
    setStatus("FINAL SOURCE completado", 1)
    return true
end

local function buildRuntimeMap()
    local lines = {
        "ROBLOX CLIENT SCANNER V1.6 - RUNTIME MAP",
        "Workspace candidates related to eggs/slots/nests/plots/bases/spawns.",
        "OWNER_MATCH means the visible path/attributes contain the local player identity; it is evidence, not server ownership proof.",
        "",
        "Total hits: " .. tostring(#state.runtimeHits),
        "",
    }
    table.sort(state.runtimeHits, function(a,b)
        if a.ownerMatch ~= b.ownerMatch then return a.ownerMatch end
        return a.path < b.path
    end)
    for index,item in ipairs(state.runtimeHits) do
        lines[#lines+1] = string.format("#%d [%s]%s", index, item.className, item.ownerMatch and " [OWNER_MATCH]" or "")
        lines[#lines+1] = "Path: " .. item.path
        lines[#lines+1] = "Parent: " .. item.parent
        if item.attrs ~= "" then lines[#lines+1] = "Attributes: " .. item.attrs end
        if item.tags ~= "" then lines[#lines+1] = "Tags: " .. item.tags end
        if item.transform ~= "" then lines[#lines+1] = "Transform: " .. item.transform end
        if item.children ~= "" then lines[#lines+1] = "Children: " .. item.children end
        lines[#lines+1] = ""
    end
    return table.concat(lines,"\n")
end
