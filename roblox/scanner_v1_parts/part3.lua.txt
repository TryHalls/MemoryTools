
local state = emptyState()

local function scannerGuiOwns(instance)
    return instance == ScreenGui or safeCall(function() return instance:IsDescendantOf(ScreenGui) end, false)
end

local function inspectInstance(instance)
    if state.seen[instance] then return end
    state.seen[instance] = true
    state.total = state.total + 1
    state.classCounts[instance.ClassName] = (state.classCounts[instance.ClassName] or 0) + 1
    local path = fullName(instance)
    if instance:IsA("RemoteEvent") or instance:IsA("RemoteFunction") then
        state.remotes[#state.remotes + 1] = {
            instance = instance,
            className = instance.ClassName,
            name = instance.Name,
            path = path,
        }
    elseif instance:IsA("ModuleScript") then
        state.modules[#state.modules + 1] = {
            instance = instance,
            className = instance.ClassName,
            name = instance.Name,
            path = path,
        }
    elseif instance:IsA("LocalScript") or instance:IsA("Script") then
        state.scripts[#state.scripts + 1] = {
            instance = instance,
            className = instance.ClassName,
            name = instance.Name,
            path = path,
        }
    end
end

local function getRoots()
    local roots = {
        { name = "ReplicatedStorage", instance = ReplicatedStorage, config = CONFIG.ROOTS.ReplicatedStorage },
        { name = "Workspace", instance = Workspace, config = CONFIG.ROOTS.Workspace },
    }
    local dynamic = {
        Character = LocalPlayer.Character,
        Backpack = LocalPlayer:FindFirstChildOfClass("Backpack"),
        PlayerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui"),
        PlayerScripts = LocalPlayer:FindFirstChild("PlayerScripts"),
    }
    for _, name in ipairs({"Character", "Backpack", "PlayerGui", "PlayerScripts"}) do
        local instance = dynamic[name]
        if instance then
            roots[#roots + 1] = { name = name, instance = instance, config = CONFIG.ROOTS[name] }
        else
            state.warnings[#state.warnings + 1] = name .. " unavailable"
        end
    end
    return roots
end

local function shouldSkipHeavy(rootName, instance)
    return rootName == "ReplicatedStorage" and instance.Parent == ReplicatedStorage and CONFIG.HEAVY_REPLICATED_TOP_LEVEL[instance.Name] == true
end

local function scanRoot(rootInfo, index, count)
    local queue = {{instance = rootInfo.instance, depth = 0}}
    local head = 1
    local scanned = 0
    local truncated = false
    terminalAdd("[SCAN] " .. rootInfo.name)
    while head <= #queue do
        if scanned >= rootInfo.config.maxObjects then
            truncated = true
            break
        end
        local node = queue[head]
        head = head + 1
        local instance = node.instance
        if not scannerGuiOwns(instance) then
            scanned = scanned + 1
            inspectInstance(instance)
            if shouldSkipHeavy(rootInfo.name, instance) then
                state.skippedHeavy[#state.skippedHeavy + 1] = fullName(instance)
            elseif node.depth < rootInfo.config.maxDepth then
                local children = safeCall(function() return instance:GetChildren() end, {})
                for _, child in ipairs(children) do
                    queue[#queue + 1] = {instance = child, depth = node.depth + 1}
                end
            end
        end
        if scanned % CONFIG.YIELD_EVERY == 0 then
            setStatus(rootInfo.name .. " • " .. tostring(scanned), (index - 1) / count)
            task.wait()
        end
    end
    state.roots[#state.roots + 1] = {name = rootInfo.name, scanned = scanned, truncated = truncated}
    if truncated then state.warnings[#state.warnings + 1] = rootInfo.name .. " reached configured scan limit" end
    terminalAdd(string.format("[%s] %s • %d", truncated and "WARN" or "OK", rootInfo.name, scanned))
    setStatus("Completado " .. rootInfo.name, index / count)
end

-- Target selection ------------------------------------------------------------

local function matchTarget(pathLower, suffix)
    return endsWith(pathLower, suffix)
end

local function allScriptRecords()
    local combined = {}
    for _, item in ipairs(state.modules) do combined[#combined + 1] = item end
    for _, item in ipairs(state.scripts) do combined[#combined + 1] = item end
    return combined
end

local function selectTargets()
    local all = allScriptRecords()
    local selected = {}
    local missing = {}
    for _, rule in ipairs(TARGET_RULES) do
        local found = nil
        for _, record in ipairs(all) do
            if matchTarget(string.lower(record.path), rule.suffix) then
                found = record
                break
            end
        end
        if found then
            selected[#selected + 1] = {label = rule.label, record = found}
        else
            missing[#missing + 1] = rule.label .. " :: " .. rule.suffix
        end
    end
    state.source.targets = selected
    state.source.missingTargets = missing
    return selected
end

-- Static decompiler analysis --------------------------------------------------

local function extractQuotedLiterals(text)
    local result = {}
    local seen = {}
    for value in string.gmatch(text, '"([^"\\]*)"') do
        local rendered = safeLiteral(value)
        if rendered ~= "" and not seen[rendered] then
            seen[rendered] = true
            result[#result + 1] = rendered
        end
    end
    for value in string.gmatch(text, "'([^'\\]*)'") do
        local rendered = safeLiteral(value)
        if rendered ~= "" and not seen[rendered] then
            seen[rendered] = true
            result[#result + 1] = rendered
        end
    end
    return result
end

local function contextBlock(lines, lineNumber)
    local from = math.max(1, lineNumber - CONFIG.CONTEXT_LINES_BEFORE)
    local to = math.min(#lines, lineNumber + CONFIG.CONTEXT_LINES_AFTER)
    local out = {}
    for index = from, to do
        out[#out + 1] = string.format("L%d: %s", index, sanitize(trim(lines[index]), CONFIG.MAX_CONTEXT_LINE))
    end
    return table.concat(out, " | ")
end

local function callReceiver(line, token)
    local escaped = token:gsub("([^%w])", "%%%1")
    local receiver = string.match(line, "([%a_][%w_]*)%s*:%s*" .. escaped)
    if receiver then return receiver end
    receiver = string.match(line, "([%a_][%w_%.%[%]\"']*)%s*%.%s*" .. escaped)
    return receiver
end

local function findAliasOrigin(lines, lineNumber, receiver)
    if not receiver or receiver == "" then return nil end
    local base = string.match(receiver, "^([%a_][%w_]*)") or receiver
    local from = math.max(1, lineNumber - CONFIG.MAX_ALIAS_LOOKBACK)
    for index = lineNumber - 1, from, -1 do
        local line = trim(lines[index])
        local lhs = string.match(line, "^local%s+([%a_][%w_]*)%s*=") or string.match(line, "^([%a_][%w_]*)%s*=")
        if lhs == base then
            return string.format("L%d: %s", index, sanitize(line, CONFIG.MAX_CONTEXT_LINE))
        end
    end
    return nil
end

local function extractRequireLines(lines)
    local out = {}
    for index, line in ipairs(lines) do
        if string.find(line, "require(", 1, true) then
            out[#out + 1] = string.format("L%d: %s", index, sanitize(trim(line), CONFIG.MAX_CONTEXT_LINE))
            if #out >= CONFIG.MAX_REQUIRES_PER_SCRIPT then break end
        end
    end
    return out
end

    local out = {}
    local seen = {}
    for index, line in ipairs(lines) do
        local lower = string.lower(line)
        local interesting = false
        for _, keyword in ipairs(INTERESTING_KEYWORDS) do
            if string.find(lower, keyword, 1, true) then
                interesting = true
                break
            end
        end
        if interesting then
            local rendered = string.format("L%d: %s", index, sanitize(trim(line), CONFIG.MAX_CONTEXT_LINE))
            if not seen[rendered] then
                seen[rendered] = true
