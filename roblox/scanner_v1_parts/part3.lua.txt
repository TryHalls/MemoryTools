                if tostring(value) == tostring(LocalPlayer.UserId) or tostring(value) == LocalPlayer.Name then
                    return true
                end
            end
        end
    end
    return false
end

local function runtimeRelevant(instance)
    local lower = string.lower(instance.Name)
    for _, term in ipairs(RUNTIME_NAME_TERMS) do
        if lower:find(term, 1, true) then return true end
    end
    local attrs = safeCall(function() return instance:GetAttributes() end, {})
    if type(attrs) == "table" then
        for key in pairs(attrs) do
            local k = string.lower(tostring(key))
            for _, term in ipairs(RUNTIME_ATTR_TERMS) do
                if k:find(term, 1, true) then return true end
            end
        end
    end
    return false
end

local function recordRuntime(instance)
    if #state.runtimeHits >= CONFIG.MAX_RUNTIME_HITS or state.runtimeHitSeen[instance] then return end
    state.runtimeHitSeen[instance] = true
    local tags = getTags(instance)
    table.sort(tags)
    local position = ""
    if instance:IsA("BasePart") then
        position = tostring(instance.CFrame)
    elseif instance:IsA("Model") then
        position = safeCall(function() return tostring(instance:GetPivot()) end, "")
    end
    state.runtimeHits[#state.runtimeHits + 1] = {
        path = fullName(instance),
        className = instance.ClassName,
        parent = instance.Parent and fullName(instance.Parent) or "nil",
        attrs = formatAttributes(instance),
        tags = table.concat(tags, ", "),
        children = childPreview(instance),
        transform = sanitize(position, 260),
        ownerMatch = pathMatchesOwner(instance),
    }
end

local function inspectInstance(instance, rootName)
    if state.seen[instance] then return end
    state.seen[instance] = true
    state.total = state.total + 1
    local path = fullName(instance)
    if instance:IsA("ModuleScript") then
        state.modules[#state.modules + 1] = {instance=instance, path=path, className=instance.ClassName}
    elseif instance:IsA("LocalScript") or instance:IsA("Script") then
        state.scripts[#state.scripts + 1] = {instance=instance, path=path, className=instance.ClassName}
    end
    if rootName == "Workspace" and runtimeRelevant(instance) then
        recordRuntime(instance)
    end
end

local function getRoots()
    local roots = {
        {name="ReplicatedStorage", instance=ReplicatedStorage, config=CONFIG.ROOTS.ReplicatedStorage},
        {name="Workspace", instance=Workspace, config=CONFIG.ROOTS.Workspace},
    }
    local playerScripts = LocalPlayer:FindFirstChild("PlayerScripts")
    if playerScripts then
        roots[#roots+1] = {name="PlayerScripts", instance=playerScripts, config=CONFIG.ROOTS.PlayerScripts}
    else
        state.warnings[#state.warnings+1] = "PlayerScripts unavailable"
    end
    if LocalPlayer.Character then
        roots[#roots+1] = {name="Character", instance=LocalPlayer.Character, config=CONFIG.ROOTS.Character}
    end
    return roots
end

local function scanRoot(info, rootIndex, rootCount)
    local queue = {{instance=info.instance, depth=0}}
    local head, scanned, truncated = 1, 0, false
    terminalAdd("[SCAN] " .. info.name)
    while head <= #queue do
        if scanned >= info.config.maxObjects then truncated = true break end
        local item = queue[head]
        head = head + 1
        local instance = item.instance
        if not ownGui(instance) then
            scanned = scanned + 1
            inspectInstance(instance, info.name)
            local skip = info.name == "ReplicatedStorage"
                and instance.Parent == ReplicatedStorage
                and CONFIG.HEAVY_REPLICATED[instance.Name]
            if not skip and item.depth < info.config.maxDepth then
                local children = safeCall(function() return instance:GetChildren() end, {})
                for _, child in ipairs(children) do
                    queue[#queue+1] = {instance=child, depth=item.depth+1}
                end
            end
        end
        if scanned % CONFIG.YIELD_EVERY == 0 then
            setStatus(info.name .. " • " .. tostring(scanned), (rootIndex-1)/rootCount)
            task.wait()
        end
    end
    state.roots[#state.roots+1] = {name=info.name, scanned=scanned, truncated=truncated}
    if truncated then state.warnings[#state.warnings+1] = info.name .. " reached configured scan limit" end
    terminalAdd(string.format("[%s] %s • %d", truncated and "WARN" or "OK", info.name, scanned))
end

local function allRecords()
    local all = {}
    for _, item in ipairs(state.modules) do all[#all+1] = item end
    for _, item in ipairs(state.scripts) do all[#all+1] = item end
    return all
end

local function selectTargets()
    local all = allRecords()
    local selected, missing = {}, {}
    for _, rule in ipairs(TARGET_RULES) do
        local found = nil
        for _, record in ipairs(all) do
            if endsWith(string.lower(record.path), rule.suffix) then
                found = record
                break
            end
        end
        if found then
            selected[#selected+1] = {label=rule.label, record=found}
        else
            missing[#missing+1] = rule.label .. " :: " .. rule.suffix
        end
    end
    state.source.targets = selected
    state.source.missing = missing
    return selected
end

-- Source analysis --------------------------------------------------------------

local function contextBlock(lines, center)
    local first = math.max(1, center - CONFIG.CONTEXT_BEFORE)
    local last = math.min(#lines, center + CONFIG.CONTEXT_AFTER)
    local out = {}
    for i=first,last do
        local marker = i == center and ">" or " "
        out[#out+1] = string.format("%s L%d: %s", marker, i, sanitize(trim(lines[i]), CONFIG.MAX_CONTEXT_LINE))
    end
    return table.concat(out, "\n"), first, last
end

local function extractRequires(lines)
    local out = {}
    for i,line in ipairs(lines) do
        if line:find("require(",1,true) then
            out[#out+1] = string.format("L%d: %s", i, sanitize(trim(line), CONFIG.MAX_CONTEXT_LINE))
            if #out >= CONFIG.MAX_REQUIRES_PER_TARGET then break end
        end
    end
    return out
end

local function focusMatches(line)
    local matches = {}
    local lower = string.lower(line)
    for _, term in ipairs(FOCUS_TERMS) do
        if lower:find(string.lower(term),1,true) then matches[#matches+1] = term end
    end
    return matches
