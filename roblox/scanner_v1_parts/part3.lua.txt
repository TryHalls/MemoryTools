    local function add(name, instance)
        local cfg = CONFIG.ROOTS[name]
        if cfg and instance then
            roots[#roots + 1] = { name = name, instance = instance, config = cfg }
        elseif cfg then
            state.warnings[#state.warnings + 1] = name .. " unavailable"
        end
    end

    add("ReplicatedStorage", ReplicatedStorage)
    add("ReplicatedFirst", ReplicatedFirst)
    add("Workspace", Workspace)

    for _, serviceName in ipairs({ "Lighting", "SoundService", "Teams" }) do
        add(serviceName, safeCall(function()
            return game:GetService(serviceName)
        end, nil))
    end

    add("Character", LocalPlayer.Character)
    add("Backpack", LocalPlayer:FindFirstChildOfClass("Backpack"))
    add("PlayerGui", LocalPlayer:FindFirstChildOfClass("PlayerGui"))
    add("PlayerScripts", LocalPlayer:FindFirstChild("PlayerScripts"))

    return roots
end

local function shouldSkipHeavyReplicatedSubtree(rootName, instance)
    return rootName == "ReplicatedStorage"
        and instance.Parent == ReplicatedStorage
        and CONFIG.HEAVY_REPLICATED_TOP_LEVEL[instance.Name] == true
end

local function scanRoot(rootInfo, rootIndex, rootCount)
    local name = rootInfo.name
    local root = rootInfo.instance
    local cfg = rootInfo.config

    terminalAdd("[SCAN] " .. name)
    setStatus("Escaneando " .. name .. "...", (rootIndex - 1) / rootCount)

    local queue = { { instance = root, depth = 0 } }
    local head = 1
    local scanned = 0
    local truncated = false
    local rootClasses = {}

    while head <= #queue do
        if scanned >= cfg.maxObjects then
            truncated = true
            break
        end

        local current = queue[head]
        head = head + 1

        local instance = current.instance
        local depth = current.depth

        if not scannerGuiOwns(instance) then
            scanned = scanned + 1
            rootClasses[instance.ClassName] = (rootClasses[instance.ClassName] or 0) + 1
            inspectInstance(instance)

            local skipHeavy = shouldSkipHeavyReplicatedSubtree(name, instance)
            if skipHeavy then
                local children = safeCall(function() return instance:GetChildren() end, {})
                state.skippedHeavy[#state.skippedHeavy + 1] = {
                    path = fullName(instance),
                    directChildren = #children,
                }
            elseif depth < cfg.maxDepth then
                local children = safeCall(function()
                    return instance:GetChildren()
                end, {})

                local limit = math.min(#children, CONFIG.MAX_CHILDREN_PER_NODE)
                if #children > CONFIG.MAX_CHILDREN_PER_NODE then
                    state.warnings[#state.warnings + 1] = string.format(
                        "%s: child limit reached at %s (%d -> %d)",
                        name,
                        fullName(instance),
                        #children,
                        CONFIG.MAX_CHILDREN_PER_NODE
                    )
                end

                for index = 1, limit do
                    if scanned + (#queue - head + 1) >= cfg.maxObjects then
                        truncated = true
                        break
                    end
                    local child = children[index]
                    if not scannerGuiOwns(child) then
                        queue[#queue + 1] = { instance = child, depth = depth + 1 }
                    end
                end
            end
        end

        if state.total % CONFIG.YIELD_EVERY == 0 then
            setStatus("Escaneando " .. name .. " • " .. tostring(scanned) .. " objetos", (rootIndex - 1) / rootCount)
            task.wait()
        end
    end

    state.roots[#state.roots + 1] = {
        name = name,
        path = fullName(root),
        scanned = scanned,
        truncated = truncated,
        classes = rootClasses,
    }

    if truncated then
        state.warnings[#state.warnings + 1] = name .. " reached configured scan limit"
        terminalAdd("[WARN] " .. name .. " truncado en " .. tostring(scanned) .. " objetos")
    else
        terminalAdd("[OK] " .. name .. " • " .. tostring(scanned) .. " objetos")
    end

    setStatus("Completado " .. name, rootIndex / rootCount)
end

local function appendMetadataLines(lines, record)
    local attrText = formatAttributes(record.attributes or {})
    local tagText = formatTags(record.tags or {})
    if attrText then
        lines[#lines + 1] = "Attributes: " .. attrText
    end
    if tagText then
        lines[#lines + 1] = "Tags: " .. tagText
    end
end

local function buildSourceCandidates()
    local ranked = {}

    for _, record in ipairs(state.modules) do
        local score = sourcePriorityScore(record.path)
        if score > 0 and record.instance then
            ranked[#ranked + 1] = { record = record, score = score }
        end
    end

    for _, record in ipairs(state.scripts) do
        if record.className == "LocalScript" and record.instance then
            local score = sourcePriorityScore(record.path)
            if score > 0 then
                ranked[#ranked + 1] = { record = record, score = score - 5 }
            end
        end
    end

    table.sort(ranked, function(a, b)
        if a.score == b.score then
            return a.record.path < b.record.path
        end
        return a.score > b.score
    end)

    local candidates = {}
    local seenPath = {}
    for _, entry in ipairs(ranked) do
        if not seenPath[entry.record.path] then
            seenPath[entry.record.path] = true
            candidates[#candidates + 1] = entry.record
            if #candidates >= CONFIG.SOURCE_MAX_CANDIDATES then
                break
            end
        end
    end

    state.source.candidates = candidates
    return candidates
end

local function countPlain(text, needle)
    local count = 0
    local cursor = 1
    while true do
        local startPos = string.find(text, needle, cursor, true)
        if not startPos then
            break
        end
        count = count + 1
        cursor = startPos + #needle
    end
    return count
end

local SOURCE_USAGE_TOKENS = {
    "FireServer",
    "InvokeServer",
    "OnClientEvent",
    "OnClientInvoke",
    "OnServerEvent",
    "OnServerInvoke",
}

local function contextFor(source, startPos, endPos)
    local radius = CONFIG.SOURCE_CONTEXT_RADIUS
    local from = math.max(1, startPos - radius)
    local to = math.min(#source, endPos + radius)
    return sanitize(string.sub(source, from, to))
end

local function nearbyUsageTokens(source, startPos, endPos)
    local from = math.max(1, startPos - 320)
    local to = math.min(#source, endPos + 320)
    local window = string.sub(source, from, to)
    local found = {}
    for _, token in ipairs(SOURCE_USAGE_TOKENS) do
        if string.find(window, token, 1, true) then
            found[#found + 1] = token
        end
    end
    return found
end

local function buildRemoteNameSet()
    local set = {}
    for _, remote in ipairs(state.remotes) do
        if string.match(remote.name, "^R[EF]/[^/]+/[^/]+$") then
            set[remote.name] = remote
        end
    end
    return set
end

local function addRemoteIndex(remoteName, path)
    local list = state.source.remoteIndex[remoteName]
    if not list then
        list = {}
        state.source.remoteIndex[remoteName] = list
    end
    for _, existing in ipairs(list) do
        if existing == path then
            return
        end
    end
    list[#list + 1] = path
end

local function analyseDecompiledRecord(record, remoteNameSet)
    state.source.attemptedCount = state.source.attemptedCount + 1

    local ok, sourceOrError = pcall(function()
        return decompile(record.instance)
    end)

    if not ok or type(sourceOrError) ~= "string" or sourceOrError == "" then
        if #state.source.failures < CONFIG.SOURCE_MAX_FAILURES then
            state.source.failures[#state.source.failures + 1] = {
                path = record.path,
                error = sanitize(sourceOrError or "decompile returned no source"),
            }
        end
        return
    end

    state.source.decompiledCount = state.source.decompiledCount + 1

    local source = sourceOrError
    local originalChars = #source
    local truncated = false
    if #source > CONFIG.SOURCE_MAX_CHARS_PER_SCRIPT then
        source = string.sub(source, 1, CONFIG.SOURCE_MAX_CHARS_PER_SCRIPT)
        truncated = true
    end
    state.source.scannedChars = state.source.scannedChars + #source

    local tokenCounts = {}
    for _, token in ipairs(SOURCE_USAGE_TOKENS) do
        local count = countPlain(source, token)
        if count > 0 then
            tokenCounts[token] = count
        end
    end

    local remoteRefs = {}
    local seenRemote = {}
    for remoteName in string.gmatch(source, "R[EF]/[%w_]+/[%w_]+") do
        if remoteNameSet[remoteName] and not seenRemote[remoteName] then
            seenRemote[remoteName] = true
            local startPos, endPos = string.find(source, remoteName, 1, true)
            if startPos then
                local ref = {
                    name = remoteName,
                    path = remoteNameSet[remoteName].path,
                    nearbyTokens = nearbyUsageTokens(source, startPos, endPos),
                    context = contextFor(source, startPos, endPos),
                }
                remoteRefs[#remoteRefs + 1] = ref
                state.source.exactRemoteReferences = state.source.exactRemoteReferences + 1
                addRemoteIndex(remoteName, record.path)
                if #remoteRefs >= CONFIG.SOURCE_MAX_REMOTE_REFS_PER_SCRIPT then
                    break
                end
            end
        end
    end

    table.sort(remoteRefs, function(a, b)
        return a.name < b.name
    end)

    state.source.results[#state.source.results + 1] = {
        className = record.className,
        path = record.path,
        sourceChars = originalChars,
        scannedChars = #source,
        truncated = truncated,
        requireCalls = countPlain(source, "require("),
        tokenCounts = tokenCounts,
        remoteRefs = remoteRefs,
    }
end

local function runSourceAnalysis()
    state.source = emptySourceState()
    state.source.attempted = true

    if not state.capabilities.decompile then
        state.source.failures[#state.source.failures + 1] = {
            path = "[executor]",
            error = "decompile() is not available",
        }
        return false, "decompile() no está disponible en este ejecutor"
    end

    local candidates = buildSourceCandidates()
    local remoteNameSet = buildRemoteNameSet()

    terminalSet("[SOURCE] V1.4\n[INFO] Decompilando solo candidatos núcleo. No se ejecutan ModuleScripts ni remotes.")

    for index, record in ipairs(candidates) do
        setStatus(
            string.format("SOURCE %d/%d • %s", index, #candidates, record.name or record.className),
            #candidates > 0 and (index - 1) / #candidates or 0
        )
        terminalAdd(string.format("[SOURCE] %d/%d %s", index, #candidates, record.path))
        analyseDecompiledRecord(record, remoteNameSet)
        task.wait()
    end

    table.sort(state.source.results, function(a, b)
        return a.path < b.path
    end)

    setStatus("SOURCE completado", 1)
    return true
end

local function buildRemoteNamespaces()
    local namespaces = {}
    for _, item in ipairs(state.remotes) do
        local namespace = string.match(item.name, "^R[EF]/([^/]+)/")
        if namespace then
            local record = namespaces[namespace]
            if not record then
                record = { RemoteEvent = 0, RemoteFunction = 0 }
                namespaces[namespace] = record
            end
            record[item.className] = (record[item.className] or 0) + 1
        end
    end
    return namespaces
end
