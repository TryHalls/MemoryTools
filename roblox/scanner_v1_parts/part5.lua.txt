    end
    return table.concat(lines,"\n")
end

local function buildReports()
    local summary = buildSummary()
    local runtime = buildRuntimeMap()
    local source = buildSourceMap()
    return {
        summary=summary,
        runtime=runtime,
        source=source,
        final=table.concat({
            summary,
            "\n\n================ RUNTIME MAP ================\n\n",
            runtime,
            "\n\n================ SOURCE MAP ================\n\n",
            source,
        }),
    }
end

local function splitText(text)
    local chunks = {}
    local cursor = 1
    while cursor <= #text do
        local finish = math.min(cursor + CONFIG.CHUNK_SIZE - 1, #text)
        if finish < #text then
            local piece = text:sub(cursor, finish)
            local last = nil
            for i=#piece,1,-1 do
                if piece:sub(i,i) == "\n" then last = i break end
            end
            if last and last > math.floor(CONFIG.CHUNK_SIZE * 0.65) then
                finish = cursor + last - 2
            end
        end
        chunks[#chunks+1] = text:sub(cursor, finish)
        cursor = finish + 1
    end
    if #chunks == 0 then chunks[1] = "" end
    return chunks
end

local function clipboard()
    if type(setclipboard) == "function" then return setclipboard end
    if type(toclipboard) == "function" then return toclipboard end
    return nil
end

local function copyChunk()
    if not activeChunks then return end
    local fn = clipboard()
    if not fn then
        terminalSet(activeChunks[activeIndex])
        setStatus("Clipboard no disponible; parte mostrada en terminal.")
        return
    end
    local ok, err = pcall(function() fn(activeChunks[activeIndex]) end)
    if ok then
        local text = string.format("%s • parte %d/%d", activeName or "Reporte", activeIndex, #activeChunks)
        terminalSet("[COPIADO]\n" .. text .. "\nCaracteres: " .. tostring(#activeChunks[activeIndex]))
        setStatus(text .. " copiada")
        NextPart.Visible = #activeChunks > 1
        if #activeChunks > 1 then NextPart.Text = string.format("SIGUIENTE PARTE (%d/%d)", activeIndex, #activeChunks) end
    else
        terminalSet("[ERROR COPY]\n" .. tostring(err))
    end
end

local function export(name,text)
    activeName = name
    activeChunks = splitText(text or "")
    activeIndex = 1
    copyChunk()
end

NextPart.MouseButton1Click:Connect(function()
    if not activeChunks or #activeChunks <= 1 then return end
    activeIndex = activeIndex + 1
    if activeIndex > #activeChunks then activeIndex = 1 end
    copyChunk()
end)
