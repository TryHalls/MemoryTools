        "Decompiler reconstruction for the two central networking modules only.",
        "No code in this export was executed by the scanner.",
        "",
    }
    for _, label in ipairs({"Networking", "Remotes"}) do
        lines[#lines + 1] = "================ " .. label .. " ================"
        local source = state.source.coreSource[label]
        if source then
            lines[#lines + 1] = source
        else
            lines[#lines + 1] = "[SOURCE NOT AVAILABLE]"
        end
        lines[#lines + 1] = ""
    end
    return table.concat(lines, "\n")
end

local function buildRemotesReport()
    local lines = {"ROBLOX CLIENT SCANNER V1.5 - REMOTES", "Total: " .. tostring(#state.remotes), ""}
    table.sort(state.remotes, function(a, b) return a.path < b.path end)
    for index, remote in ipairs(state.remotes) do
        lines[#lines + 1] = string.format("#%d [%s] %s", index, remote.className, remote.path)
    end
    return table.concat(lines, "\n")
end

local function buildReports()
    return {
        summary = buildSummary(),
        network = buildNetworkMap(),
        coreSource = buildCoreSource(),
        remotes = buildRemotesReport(),
        targets = buildTargetsReport(),
    }
end

local function splitText(text, maxLength)
    local chunks = {}
    local cursor = 1
    while cursor <= #text do
        local finish = math.min(cursor + maxLength - 1, #text)
        if finish < #text then
            local candidate = string.sub(text, cursor, finish)
            local last = nil
            for index = #candidate, 1, -1 do
                if string.sub(candidate, index, index) == "\n" then last = index break end
            end
            if last and last > math.floor(maxLength * 0.65) then finish = cursor + last - 2 end
        end
        chunks[#chunks + 1] = string.sub(text, cursor, finish)
        cursor = finish + 1
    end
    if #chunks == 0 then chunks[1] = "" end
    return chunks
end

local function clipboardFunction()
    if type(setclipboard) == "function" then return setclipboard end
    if type(toclipboard) == "function" then return toclipboard end
    return nil
end

local function copyActiveChunk()
    if not activeChunks then return end
    local copy = clipboardFunction()
    if not copy then
        terminalSet(activeChunks[activeChunkIndex])
        setStatus("Portapapeles no disponible; bloque mostrado en terminal.")
        return
    end
    local ok, err = pcall(function() copy(activeChunks[activeChunkIndex]) end)
    if ok then
        local label = string.format("%s • parte %d/%d", activeExportName or "Reporte", activeChunkIndex, #activeChunks)
        setStatus(label .. " copiada")
        terminalSet("[COPIADO]\n" .. label .. "\nCaracteres: " .. tostring(#activeChunks[activeChunkIndex]))
        NextPart.Visible = #activeChunks > 1
        if #activeChunks > 1 then NextPart.Text = string.format("SIGUIENTE PARTE (%d/%d)", activeChunkIndex, #activeChunks) end
    else
        terminalSet("[ERROR COPY]\n" .. tostring(err))
    end
end

local function exportCategory(name, text)
    activeExportName = name
    activeChunks = splitText(text or "", CONFIG.CHUNK_SIZE)
    activeChunkIndex = 1
    copyActiveChunk()
end

NextPart.MouseButton1Click:Connect(function()
    if not activeChunks or #activeChunks <= 1 then return end
    activeChunkIndex = activeChunkIndex + 1
    if activeChunkIndex > #activeChunks then activeChunkIndex = 1 end
    copyActiveChunk()
end)
