local function safeCall(callback, fallback)
    local ok, result = pcall(callback)
    if ok then
        return result
    end
    return fallback
end

local function sanitize(value)
    local text = tostring(value)
    text = text:gsub("\r", "\\r"):gsub("\n", "\\n"):gsub("\t", "\\t")
    if #text > CONFIG.MAX_STRING then
        return text:sub(1, CONFIG.MAX_STRING) .. "...[TRUNCATED]"
    end
    return text
end

local function fullName(instance)
    return safeCall(function()
        return instance:GetFullName()
    end, instance and instance.Name or "unknown")
end

local function getAttributes(instance)
    local value = safeCall(function()
        return instance:GetAttributes()
    end, {})
    return type(value) == "table" and value or {}
end

local function getTags(instance)
    local value = safeCall(function()
        return CollectionService:GetTags(instance)
    end, {})
    return type(value) == "table" and value or {}
end

local function sortedKeys(tbl)
    local result = {}
    for key in pairs(tbl) do
        result[#result + 1] = key
    end
    table.sort(result, function(a, b)
        return tostring(a) < tostring(b)
    end)
    return result
end

local function serialize(value)
    local kind = typeof(value)
    if value == nil then
        return "nil"
    elseif kind == "string" then
        return '"' .. sanitize(value) .. '"'
    elseif kind == "number" or kind == "boolean" then
        return tostring(value)
    elseif kind == "Vector3" then
        return string.format("Vector3(%.2f, %.2f, %.2f)", value.X, value.Y, value.Z)
    elseif kind == "Vector2" then
        return string.format("Vector2(%.2f, %.2f)", value.X, value.Y)
    elseif kind == "CFrame" then
        local p = value.Position
        return string.format("CFrame(%.2f, %.2f, %.2f)", p.X, p.Y, p.Z)
    elseif kind == "Color3" then
        return string.format("Color3(%.3f, %.3f, %.3f)", value.R, value.G, value.B)
    elseif kind == "EnumItem" then
        return tostring(value)
    elseif kind == "Instance" then
        return fullName(value)
    end
    return sanitize(value)
end

local function hasPriorityWord(instance)
    local name = string.lower(instance.Name or "")
    for _, word in ipairs(PRIORITY_WORDS) do
        if string.find(name, word, 1, true) then
            return true
        end
    end
    return false
end

local function isCoreModulePath(path)
    local lower = string.lower(path)
    for _, pattern in ipairs(CORE_PATTERNS) do
        if string.find(lower, pattern, 1, true) then
            return true
        end
    end
    if string.find(lower, ".playerscripts.game.", 1, true) then
        return true
    end
    return false
end

local function isSensitiveAttributeKey(key)
    local lower = string.lower(tostring(key))
    for _, word in ipairs({ "apikey", "api_key", "secret", "token", "password", "credential", "authkey", "accesskey" }) do
        if string.find(lower, word, 1, true) then
            return true
        end
    end
    return false
end

local function formatAttributes(attributes)
    if next(attributes) == nil then
        return nil
    end
    local parts = {}
    for _, key in ipairs(sortedKeys(attributes)) do
        local valueText = isSensitiveAttributeKey(key) and "[REDACTED]" or serialize(attributes[key])
        parts[#parts + 1] = tostring(key) .. "=" .. valueText
    end
    return table.concat(parts, "; ")
end

local function formatTags(tags)
    if #tags == 0 then
        return nil
    end
    table.sort(tags)
    return table.concat(tags, ", ")
end
