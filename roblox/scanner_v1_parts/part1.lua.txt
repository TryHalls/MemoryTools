local function safeCall(callback, fallback)
    local ok, result = pcall(callback)
    if ok then
        return result
    end
    return fallback
end

local function fullName(instance)
    return safeCall(function()
        return instance:GetFullName()
    end, instance and instance.Name or "unknown")
end

local function sanitize(value, limit)
    local text = tostring(value)
    text = text:gsub("\r", "\\r"):gsub("\n", "\\n"):gsub("\t", "\\t")
    local maximum = limit or CONFIG.MAX_STRING
    if #text > maximum then
        return string.sub(text, 1, maximum) .. "...[TRUNCATED]"
    end
    return text
end

local function trim(text)
    text = tostring(text or "")
    text = text:gsub("^%s+", "")
    text = text:gsub("%s+$", "")
    return text
end

local function endsWith(text, suffix)
    if #suffix > #text then
        return false
    end
    return string.sub(text, #text - #suffix + 1) == suffix
end

local function splitLines(source)
    local lines = {}
    source = tostring(source or "")
    if string.sub(source, -1) ~= "\n" then
        source = source .. "\n"
    end
    for line in string.gmatch(source, "(.-)\n") do
        lines[#lines + 1] = line
    end
    return lines
end

local function isSensitiveLiteral(value)
    if #value < 28 then
        return false
    end
    if string.find(value, "%s") then
        return false
    end
    local compact = value:gsub("[%w%-%_%.]", "")
    return #compact <= math.max(2, math.floor(#value * 0.08))
end

local function safeLiteral(value)
    if isSensitiveLiteral(value) then
        return "[REDACTED_LONG_LITERAL]"
    end
    return sanitize(value, 120)
end

local function getAttributes(instance)
    local attrs = safeCall(function() return instance:GetAttributes() end, {})
    return type(attrs) == "table" and attrs or {}
end

local function getTags(instance)
    local tags = safeCall(function() return CollectionService:GetTags(instance) end, {})
    return type(tags) == "table" and tags or {}
end

local function sensitiveAttributeKey(key)
    local lower = string.lower(tostring(key))
    for _, word in ipairs({"apikey", "api_key", "secret", "token", "password", "credential", "authkey", "accesskey"}) do
        if string.find(lower, word, 1, true) then
            return true
        end
    end
    return false
end

local function formatMetadata(instance)
    local attrs = getAttributes(instance)
    local tags = getTags(instance)
    local parts = {}
    for key, value in pairs(attrs) do
        local rendered = sensitiveAttributeKey(key) and "[REDACTED]" or sanitize(value, 120)
        parts[#parts + 1] = tostring(key) .. "=" .. rendered
    end
    table.sort(parts)
    if #tags > 0 then
        table.sort(tags)
        parts[#parts + 1] = "Tags=" .. table.concat(tags, ",")
    end
    return table.concat(parts, "; ")
end
