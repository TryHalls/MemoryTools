local function safeCall(fn, fallback)
    local ok, result = pcall(fn)
    if ok then return result end
    return fallback
end

local function trim(text)
    text = tostring(text or "")
    return (text:gsub("^%s+", ""):gsub("%s+$", ""))
end

local function sanitize(value, limit)
    local text = tostring(value or "")
    text = text:gsub("\r", "\\r"):gsub("\n", "\\n"):gsub("\t", "\\t")
    limit = limit or CONFIG.MAX_CONTEXT_LINE
    if #text > limit then
        return text:sub(1, limit) .. "...[TRUNCATED]"
    end
    return text
end

local function fullName(instance)
    return safeCall(function() return instance:GetFullName() end, instance and instance.Name or "unknown")
end

local function endsWith(text, suffix)
    if #suffix > #text then return false end
    return text:sub(#text - #suffix + 1) == suffix
end

local function splitLines(source)
    source = tostring(source or ""):gsub("\r\n", "\n"):gsub("\r", "\n")
    if source:sub(-1) ~= "\n" then source = source .. "\n" end
    local lines = {}
    for line in source:gmatch("(.-)\n") do
        lines[#lines + 1] = line
    end
    return lines
end

local function getTags(instance)
    local tags = safeCall(function() return CollectionService:GetTags(instance) end, {})
    return type(tags) == "table" and tags or {}
end

local function sensitiveKey(key)
    local lower = string.lower(tostring(key))
    for _, word in ipairs({"apikey", "api_key", "secret", "token", "password", "credential", "authkey", "accesskey"}) do
        if lower:find(word, 1, true) then return true end
    end
    return false
end

local function formatAttributes(instance)
    local attrs = safeCall(function() return instance:GetAttributes() end, {})
    if type(attrs) ~= "table" then return "" end
    local keys = {}
    for key in pairs(attrs) do keys[#keys + 1] = key end
    table.sort(keys, function(a, b) return tostring(a) < tostring(b) end)
    local out = {}
    for _, key in ipairs(keys) do
        local value = sensitiveKey(key) and "[REDACTED]" or sanitize(attrs[key], 160)
        out[#out + 1] = tostring(key) .. "=" .. value
    end
    return table.concat(out, "; ")
end

local function childPreview(instance)
    local children = safeCall(function() return instance:GetChildren() end, {})
    local names = {}
    for index, child in ipairs(children) do
        if index > CONFIG.MAX_CHILD_PREVIEW then break end
        names[#names + 1] = child.Name .. "[" .. child.ClassName .. "]"
    end
    local suffix = #children > CONFIG.MAX_CHILD_PREVIEW and (" ... +" .. tostring(#children - CONFIG.MAX_CHILD_PREVIEW)) or ""
    return table.concat(names, ", ") .. suffix
end
