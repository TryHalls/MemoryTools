-- Drag support for mouse/touch.
local dragging = false
local dragStart
local startPosition

Topbar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPosition = Main.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(
            startPosition.X.Scale,
            startPosition.X.Offset + delta.X,
            startPosition.Y.Scale,
            startPosition.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

Close.MouseButton1Click:Connect(function()
    ScreenGui:Destroy()
end)

local function emptyState()
    return {
        total = 0,
        roots = {},
        remotes = {},
        modules = {},
        scripts = {},
        structure = {},
        metadata = {},
        values = {},
        tools = {},
        interactions = {},
        gui = {},
        priority = {},
        warnings = {},
        classCounts = {},
        limits = {
            structure = false,
            metadata = false,
            values = false,
            gui = false,
            priority = false,
        },
    }
end

local state = emptyState()

local function scannerGuiOwns(instance)
    return instance == ScreenGui or safeCall(function()
        return instance:IsDescendantOf(ScreenGui)
    end, false)
end

local function incrementClass(className)
    state.classCounts[className] = (state.classCounts[className] or 0) + 1
end

local function addLimited(list, item, limitName, maximum)
    if #list < maximum then
        list[#list + 1] = item
    else
        state.limits[limitName] = true
    end
end

local function makeCommonRecord(instance)
    return {
        className = instance.ClassName,
        name = instance.Name,
        path = fullName(instance),
        attributes = getAttributes(instance),
        tags = getTags(instance),
    }
end

local function inspectInstance(instance)
    state.total = state.total + 1
    incrementClass(instance.ClassName)

    local attributes = getAttributes(instance)
    local tags = getTags(instance)
    local path = fullName(instance)

    if instance:IsA("RemoteEvent") or instance:IsA("RemoteFunction") then
        state.remotes[#state.remotes + 1] = {
            className = instance.ClassName,
            name = instance.Name,
            path = path,
            parent = instance.Parent and fullName(instance.Parent) or "nil",
            attributes = attributes,
            tags = tags,
        }
    elseif instance:IsA("ModuleScript") then
        state.modules[#state.modules + 1] = {
            className = instance.ClassName,
            name = instance.Name,
            path = path,
            attributes = attributes,
            tags = tags,
        }
    elseif instance:IsA("LocalScript") or instance:IsA("Script") then
        state.scripts[#state.scripts + 1] = {
            className = instance.ClassName,
            name = instance.Name,
            path = path,
            enabled = safeCall(function()
                return instance.Enabled
            end, "unavailable"),
            attributes = attributes,
            tags = tags,
        }
    end

    if instance:IsA("Folder") or instance:IsA("Configuration") then
        local children = safeCall(function()
            return instance:GetChildren()
        end, {})
        local childClasses = {}
        for _, child in ipairs(children) do
            childClasses[child.ClassName] = (childClasses[child.ClassName] or 0) + 1
        end
        addLimited(state.structure, {
            className = instance.ClassName,
            path = path,
            childCount = #children,
            childClasses = childClasses,
        }, "structure", CONFIG.MAX_STRUCTURE_ENTRIES)
    end

    if next(attributes) ~= nil or #tags > 0 then
        addLimited(state.metadata, {
            className = instance.ClassName,
            path = path,
            attributes = attributes,
            tags = tags,
        }, "metadata", CONFIG.MAX_METADATA_ENTRIES)
    end

    if instance:IsA("ValueBase") then
        addLimited(state.values, {
            className = instance.ClassName,
            path = path,
            value = safeCall(function()
                return instance.Value
            end, "unavailable"),
        }, "values", CONFIG.MAX_VALUE_ENTRIES)
    end

    if instance:IsA("Tool") then
        state.tools[#state.tools + 1] = {
            path = path,
            tooltip = safeCall(function() return instance.ToolTip end, ""),
            requiresHandle = safeCall(function() return instance.RequiresHandle end, "unavailable"),
            canBeDropped = safeCall(function() return instance.CanBeDropped end, "unavailable"),
            attributes = attributes,
            tags = tags,
        }
    end

    if instance:IsA("ProximityPrompt") then
        state.interactions[#state.interactions + 1] = {
            className = instance.ClassName,
            path = path,
            actionText = safeCall(function() return instance.ActionText end, ""),
            objectText = safeCall(function() return instance.ObjectText end, ""),
            maxDistance = safeCall(function() return instance.MaxActivationDistance end, "unavailable"),
            holdDuration = safeCall(function() return instance.HoldDuration end, "unavailable"),
        }
    elseif instance:IsA("ClickDetector") then
        state.interactions[#state.interactions + 1] = {
            className = instance.ClassName,
            path = path,
            maxDistance = safeCall(function() return instance.MaxActivationDistance end, "unavailable"),
        }
    end

    if instance:IsA("ScreenGui") or instance:IsA("TextButton") or instance:IsA("ImageButton") or instance:IsA("TextBox") then
        addLimited(state.gui, {
            className = instance.ClassName,
            path = path,
            text = (instance:IsA("TextButton") or instance:IsA("TextBox")) and safeCall(function() return instance.Text end, "") or nil,
        }, "gui", CONFIG.MAX_GUI_ENTRIES)
    end

    if hasPriorityWord(instance) then
        addLimited(state.priority, {
            className = instance.ClassName,
            path = path,
        }, "priority", CONFIG.MAX_KEYWORD_HITS)
    end
end
